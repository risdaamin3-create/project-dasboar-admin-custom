// --- SISTEM KEAMANAN & LOGIN BARU ---
function hashPassword(password) {
  if(!password) return "";
  var rawHash = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, password.toString());
  var txtHash = '';
  for (var j = 0; j < rawHash.length; j++) {
    var hashVal = rawHash[j];
    if (hashVal < 0) hashVal += 256;
    if (hashVal.toString(16).length == 1) txtHash += "0";
    txtHash += hashVal.toString(16);
  }
  return txtHash;
}

function verifyLogin(username, password) {
  try {
    const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName('Users');
    if(!sheet || sheet.getLastRow() < 2) return { success: false, message: 'Database User tidak ditemukan!' };
    
    // Sesuaikan pembacaan kolom agar dinamis dengan struktur database baru Anda
    const headers = sheet.getRange(1, 1, 1, Math.max(sheet.getLastColumn(), 5)).getValues()[0];
    const userIdx = headers.indexOf('Username');
    const passIdx = headers.indexOf('Password');
    const nameIdx = headers.indexOf('Nama');
    const roleIdx = headers.indexOf('Role');
    const photoIdx = headers.indexOf('Foto Profil');
    
    if(userIdx === -1 || passIdx === -1) {
      return { success: false, message: 'Format tabel Users salah (Kolom Username/Password hilang)!' };
    }
    
    const data = sheet.getRange(2, 1, sheet.getLastRow() - 1, Math.max(sheet.getLastColumn(), 5)).getValues();
    const inputHash = hashPassword(password);
    
    for (let i = 0; i < data.length; i++) {
      let dbUser = data[i][userIdx] ? data[i][userIdx].toString().trim() : '';
      let dbPass = data[i][passIdx] ? data[i][passIdx].toString().trim() : '';
      
      // Verifikasi menggunakan Hash atau Teks Biasa
      if (username === dbUser && (inputHash === dbPass || password === dbPass)) {
        
        // Auto-upgrade: Ubah password biasa menjadi hash di Spreadsheet saat login
        if (password === dbPass && password !== "") {
          sheet.getRange(i + 2, passIdx + 1).setValue(inputHash);
        }
        
        let dbName = (nameIdx !== -1 && data[i][nameIdx]) ? data[i][nameIdx].toString().trim() : dbUser;
        let dbRole = (roleIdx !== -1 && data[i][roleIdx]) ? data[i][roleIdx].toString().trim() : 'Staff';
        let dbFoto = (photoIdx !== -1 && data[i][photoIdx]) ? data[i][photoIdx].toString().trim() : '';
        
        logActivity('Sistem Keamanan', 'Login berhasil', dbName);
        return { 
          success: true, 
          nama: dbName, 
          username: dbUser, 
          role: dbRole, 
          foto: dbFoto 
        };
      }
    }
    
    logActivity('Sistem Keamanan', 'Gagal login (Username/Password salah)', username);
    return { success: false, message: 'Username atau Password salah!' };
  } catch (e) {
    return { success: false, message: 'Error Server: ' + e.toString() };
  }
}

function getDashboardStats() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const productsSheet = ss.getSheetByName('Products');
  let prodCount = productsSheet ? Math.max(0, productsSheet.getLastRow() - 1) : 0;
  
  // Mapping HPP
  let hppMap = {};
  if (prodCount > 0) {
      let pData = productsSheet.getDataRange().getValues();
      for(let i=1; i<pData.length; i++) {
          let pName = pData[i][1];
          let pHPP = parseFloat(pData[i][24]) || 0; // HPP ada di index 24
          if(pName && pHPP) { hppMap[pName.toString().toLowerCase().trim()] = pHPP; }
      }
  }

  let postCount = 0;
  const config = getSystemConfig();
  if (config.blogId) {
    try {
      const options = { "method": "get", "headers": { "Authorization": "Bearer " + ScriptApp.getOAuthToken() }, "muteHttpExceptions": true };
      const response = UrlFetchApp.fetch("https://www.googleapis.com/blogger/v3/blogs/" + config.blogId + "/posts?maxResults=500&fetchBodies=false&view=ADMIN", options);
      const json = JSON.parse(response.getContentText());
      if (json.items) { postCount = json.items.length; }
    } catch(e) {}
  }
  
  let gross = 0; let net = 0;
  const ordersSheet = ss.getSheetByName('Orders');
  if (ordersSheet && ordersSheet.getLastRow() > 1) {
     const orderData = ordersSheet.getDataRange().getValues();
     for (let i = 1; i < orderData.length; i++) {
        if (orderData[i][8] === 'Selesai') {
           let amount = parseFloat(orderData[i][6]) || 0;
           gross += amount;
           let itemsJson = orderData[i][5];
           let orderHppTotal = 0;
           try {
               let items = JSON.parse(itemsJson);
               if(Array.isArray(items)) {
                   items.forEach(it => {
                       let itName = it.name ? it.name.toString().toLowerCase().trim() : '';
                       let hpp = hppMap[itName] || 0;
                       orderHppTotal += hpp * (parseFloat(it.qty) || 1);
                   });
               }
           } catch(e) {}
           net += (amount - orderHppTotal);
        }
     }
  }
  
  return { products: prodCount, posts: postCount, gross: gross, net: net };
}

function callGeminiAI(prompt, isHtmlMode) {
  const config = getSystemConfig();
  if (!config.geminiApi) throw new Error("API Key Gemini belum diatur di menu Pengaturan!");

  const cleanApiKey = String(config.geminiApi).trim();
  
  // PERUBAHAN DI SINI: Menyesuaikan persis dengan cURL Google AI Studio Anda
  const url = "https://generativelanguage.googleapis.com/v1beta/models/gemini-flash-latest:generateContent?key=" + cleanApiKey;
  
  let finalPrompt = prompt;
  if (isHtmlMode) {
    // Menerapkan aturan output teks biasa tanpa HTML untuk aksesibilitas
    finalPrompt += "\n\nPENTING: Tuliskan respons dalam format teks biasa (plaintext) murni tanpa menggunakan tag HTML.";
  }

  const payload = { "contents": [{ "parts": [{"text": finalPrompt}] }] };
  const options = { 
    "method": "post", 
    "contentType": "application/json", 
    "payload": JSON.stringify(payload), 
    "muteHttpExceptions": true 
  };

  try {
    const response = UrlFetchApp.fetch(url, options);
    const json = JSON.parse(response.getContentText());
    
    if (json.error) throw new Error("Server AI: " + json.error.message);

    if (!json.candidates || json.candidates.length === 0) {
      throw new Error("Respons kosong atau diblokir oleh sistem keamanan Google.");
    }
    if (!json.candidates[0].content || !json.candidates[0].content.parts) {
      throw new Error("Format respons ditolak. Status: " + json.candidates[0].finishReason);
    }

    let text = json.candidates[0].content.parts[0].text;
    if (isHtmlMode) {
      // Menggunakan hex codes menghindari error backtick saat copy-paste
      text = text.replace(/\x60\x60\x60html/gi, "").replace(/\x60\x60\x60/gi, "").trim();
    }
    return text;
  } catch (e) {
    throw new Error(e.message);
  }
}

function uploadSingleImageToDrive(base64Str) {
  const config = getSystemConfig();
  if(!config.driveId) throw new Error("ID Folder Drive belum diisi di menu Pengaturan!");
  try {
    const parts = base64Str.split(','); const mimeType = parts[0].split(':')[1].split(';')[0];
    const cleanData = parts[1].replace(/[^A-Za-z0-9+/=]/g, '');
    const blob = Utilities.newBlob(Utilities.base64Decode(cleanData), mimeType, 'IMG_BLOG_' + Date.now() + '_' + Math.floor(Math.random()*1000) + '.jpg');
    const fileMetadata = { name: blob.getName(), parents: [config.driveId] };
    const file = Drive.Files.create(fileMetadata, blob);
    Drive.Permissions.create({ role: 'reader', type: 'anyone' }, file.id);
    return "https://drive.google.com/uc?id=" + file.id;
  } catch(e) { throw new Error(e.message); }
}

function processHtmlImages(htmlContent) {
  if (!htmlContent) return '';
  let updated = htmlContent.replace(/https:\/\/drive\.google\.com\/uc\?id=([a-zA-Z0-9_-]+)/g, 'https://drive.google.com/thumbnail?id=$1&sz=w1000');
  updated = updated.replace(/https:\/\/lh3\.googleusercontent\.com\/d\/([a-zA-Z0-9_-]+)/g, 'https://drive.google.com/thumbnail?id=$1&sz=w1000');
  return updated;
}

function deleteDriveImagesFromHtml(htmlContent) {
  if (!htmlContent) return;
  const regex = /(?:thumbnail\?id=|uc\?id=|d\/)([a-zA-Z0-9_-]{25,})/g;
  let match; let idsToDelete = [];
  while ((match = regex.exec(htmlContent)) !== null) { idsToDelete.push(match[1]); }
  idsToDelete = [...new Set(idsToDelete)];
  idsToDelete.forEach(id => { try { Drive.Files.remove(id); } catch(e) {} });
}

function processImagesToDrive(params) {
  const config = getSystemConfig();
  if(!config.driveId) throw new Error("ID Folder Drive belum diisi di Pengaturan!");

  function convertBase64ToDriveUrl(base64Str) {
    if (!base64Str || typeof base64Str !== 'string') return base64Str;
    if (base64Str.startsWith('http')) return base64Str; 
    if (!base64Str.startsWith('data:image')) return base64Str; 
    try {
      const parts = base64Str.split(','); const mimeType = parts[0].split(':')[1].split(';')[0];
      const cleanData = parts[1].replace(/[^A-Za-z0-9+/=]/g, '');
      const blob = Utilities.newBlob(Utilities.base64Decode(cleanData), mimeType, 'IMG_' + Date.now() + '_' + Math.floor(Math.random()*1000) + '.jpg');
      const fileMetadata = { name: blob.getName(), parents: [config.driveId] };
      const file = Drive.Files.create(fileMetadata, blob);
      Drive.Permissions.create({ role: 'reader', type: 'anyone' }, file.id);
      return "https://drive.google.com/uc?id=" + file.id;
    } catch(e) { throw new Error("ADVANCED DRIVE ERROR: " + e.message); }
  }

  if (params.img) {
    try {
      let parsed = JSON.parse(params.img);
      if (Array.isArray(parsed)) { for (let i = 0; i < parsed.length; i++) { parsed[i] = convertBase64ToDriveUrl(parsed[i]); } params.img = JSON.stringify(parsed); } 
      else { params.img = JSON.stringify([convertBase64ToDriveUrl(parsed)]); }
    } catch(e) { if (params.img.startsWith('data:image')) { params.img = JSON.stringify([convertBase64ToDriveUrl(params.img)]); } }
  }
  if (params.variant) {
    try {
      let varsArray = JSON.parse(params.variant);
      if (Array.isArray(varsArray)) { for (let i = 0; i < varsArray.length; i++) { if (varsArray[i].image) { varsArray[i].image = convertBase64ToDriveUrl(varsArray[i].image); } } params.variant = JSON.stringify(varsArray); }
    } catch(e) {}
  }
  return params;
}

function buildProductJSONLD(p) {
  let images = [];
  try { let parsedImg = JSON.parse(p.img); if (Array.isArray(parsedImg)) { images = parsedImg; } else { images.push(parsedImg); } } 
  catch(e) { if(p.img) images.push(p.img); }

  let availabilitySchema = "https://schema.org/InStock";
  if (p.availability && p.availability.includes("OutOfStock")) availabilitySchema = "https://schema.org/OutOfStock";
  if (p.availability && p.availability.includes("PreOrder")) availabilitySchema = "https://schema.org/PreOrder";

  let baseUrl = "https://example.com";
  try { 
      let settings = getSettingsData(); 
      if (settings.storeUrl) baseUrl = settings.storeUrl.replace(/\/$/, ""); 
  } catch(e) {}
  
  let productUrl = p.canonical ? p.canonical : baseUrl + "/p/" + p.slug + ".html";

  let schema = {
    "@context": "https://schema.org/", "@type": "Product", "name": p.name, "image": images, "description": p.desc ? p.desc.substring(0, 200) : "", "sku": p.sku || "",
    "offers": { "@type": "Offer", "url": productUrl, "priceCurrency": "IDR", "price": p.price, "availability": availabilitySchema, "itemCondition": "https://schema.org/NewCondition" }
  };

  // Tambahkan aggregateRating dari Testimonials jika ada
  if (p.sku) {
     try {
         var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
         var testSheet = ss.getSheetByName('Testimonials');
         if (testSheet) {
             var tData = testSheet.getDataRange().getValues();
             var ratingSum = 0; var ratingCount = 0;
             for (var i = 1; i < tData.length; i++) {
                 if (tData[i][3] == p.sku) {
                     ratingSum += parseInt(tData[i][4]);
                     ratingCount++;
                 }
             }
             if (ratingCount > 0) {
                 schema.aggregateRating = {
                     "@type": "AggregateRating",
                     "ratingValue": (ratingSum / ratingCount).toFixed(1),
                     "reviewCount": ratingCount
                 };
             }
         }
     } catch(e) {}
  }

  if (p.youtube) {
    let ytMatch = p.youtube.match(/(?:v=|\/)([a-zA-Z0-9_-]{11})/);
    if(ytMatch) {
       schema.video = {
          "@type": "VideoObject",
          "name": "Video: " + p.name,
          "description": p.desc ? p.desc.substring(0, 200) : "",
          "thumbnailUrl": "https://img.youtube.com/vi/" + ytMatch[1] + "/maxresdefault.jpg",
          "uploadDate": new Date().toISOString(),
          "embedUrl": "https://www.youtube.com/embed/" + ytMatch[1]
       };
    }
  }

  if (p.faq) {
    try {
      let faqData = JSON.parse(p.faq);
      if (faqData.length > 0) { schema.mainEntity = faqData.map(f => ({ "@type": "Question", "name": f.q, "acceptedAnswer": { "@type": "Answer", "text": f.a } })); }
    } catch(e) {}
  }
  return '<script type="application/ld+json">\n' + JSON.stringify(schema) + '\n</script>';
}

function getSafeImgUrl(url) {
  if(!url) return '';
  let match = url.match(/id=([a-zA-Z0-9_-]+)/); if (!match) match = url.match(/d\/([a-zA-Z0-9_-]+)/);
  if(match) { return 'https://drive.google.com/thumbnail?id=' + match[1] + '&sz=w1000'; }
  return url;
}

function publishBlogPost(title, content, labels, meta, postDate, location, permalink, isDraft, canonical) {
  const config = getSystemConfig();
  if(!config.blogId) return "Error: Blog ID belum diatur di Pengaturan!";

  let schemaArticle = { "@context": "https://schema.org", "@type": "Article", "headline": title, "description": meta, "datePublished": postDate || new Date().toISOString() };
  let schemaBreadcrumb = { "@context": "https://schema.org", "@type": "BreadcrumbList", "itemListElement": [ { "@type": "ListItem", "position": 1, "name": "Artikel" }, { "@type": "ListItem", "position": 2, "name": title } ] };
  
  let safeContent = processHtmlImages(content);
  let finalContent = safeContent + '\n<script type="application/ld+json">\n' + JSON.stringify(schemaArticle) + '\n</script>\n<script type="application/ld+json">\n' + JSON.stringify(schemaBreadcrumb) + '\n</script>';
  if (canonical) { finalContent += '\n<link rel="canonical" href="' + canonical + '">'; }

  const payload = { "kind": "blogger#post", "title": title, "content": finalContent };
  if(labels) payload.labels = labels.split(',').map(l => l.trim());
  if(meta) payload.customMetaData = meta; if(location) payload.location = { "name": location };
  if(postDate) { try { payload.published = new Date(postDate).toISOString(); } catch(e) {} }

  const options = { "method": "post", "headers": { "Authorization": "Bearer " + ScriptApp.getOAuthToken() }, "contentType": "application/json", "payload": JSON.stringify(payload), "muteHttpExceptions": true };
  try {
    const draftParam = isDraft ? "?isDraft=true" : "";
    const response = UrlFetchApp.fetch("https://www.googleapis.com/blogger/v3/blogs/" + config.blogId + "/posts/" + draftParam, options);
    const json = JSON.parse(response.getContentText());
    if (json.error) return "Error API: " + json.error.message;
    return isDraft ? "Artikel disimpan sebagai DRAF." : "Artikel SEO berhasil TERBIT.";
  } catch(e) { return "Error sistem: " + e.message; }
}

function exportProductsCSV() {
  const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName('Products');
  if(!sheet) return "";
  const data = sheet.getDataRange().getValues();
  
  let csvArr = [];
  // Header CSV
  csvArr.push(["Row_ID", "Nama_Produk", "SKU", "Harga_Normal", "Harga_Diskon", "Stok_Induk"]);
  
  // Kita mulai dari baris ke-2 (index 1) karena baris ke-1 adalah header di Spreadsheet
  for (let i = 1; i < data.length; i++) {
    let rowIdx = i + 1; // ID baris yang sesungguhnya di database
    let name = String(data[i][1]).replace(/"/g, '""'); // Membersihkan tanda kutip agar aman
    let sku = String(data[i][9]).replace(/"/g, '""');
    let price = data[i][2];
    let disc = data[i][14] || "";
    let stock = data[i][10] || 0;
    
    // Gabungkan dengan koma
    csvArr.push([rowIdx, `"${name}"`, `"${sku}"`, price, disc, stock].join(","));
  }
  return csvArr.join("\n");
}

function importProductsCSV(csvString) {
  try {
    const data = Utilities.parseCsv(csvString);
    const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName('Products');
    let count = 0;
    
    // Loop dari baris ke-2 CSV (mengabaikan header)
    for (let i = 1; i < data.length; i++) {
      let row = data[i];
      if(row.length < 6) continue;
      
      let rowIdx = parseInt(row[0]);
      let name = row[1];
      let sku = row[2];
      let price = row[3];
      let disc = row[4];
      let stock = row[5];
      
      // Pastikan Row_ID valid untuk menghindari penimpaan data yang salah
      if(rowIdx > 1 && rowIdx <= sheet.getLastRow()) {
        sheet.getRange(rowIdx, 2).setValue(name);     // Update Nama
        sheet.getRange(rowIdx, 3).setValue(price);    // Update Harga
        sheet.getRange(rowIdx, 10).setValue(sku);     // Update SKU
        sheet.getRange(rowIdx, 11).setValue(stock);   // Update Stok
        sheet.getRange(rowIdx, 15).setValue(disc);    // Update Diskon
        count++;
      }
    }
    
    logActivity('Bulk Edit CSV', 'Memperbarui massal ' + count + ' produk via CSV', 'Admin');
    return "✅ Berhasil memperbarui " + count + " produk secara massal!";
  } catch(e) {
    throw new Error("Gagal mengimpor CSV. Pastikan Anda tidak menghapus kolom Row_ID. Detail: " + e.message);
  }
}

    INSTRUKSI FORMAT SANGAT KETAT:
    1. Seluruh artikel WAJIB ditulis murni dalam format Plaintext.
    2. DILARANG KERAS menyertakan tag pemformatan HTML, XML, atau blok kode apa pun.
    3. Teks harus siap disalin, copy-friendly, dan sangat kompatibel untuk diekstraksi ke sistem NotebookLM Google.
    Awali tulisan dengan "JUDUL ARTIKEL: [Judul yang relevan]"`;
    
    try {
      let aiText = callGeminiAI(prompt, true); 
      // Menyimpan hasil ke dalam array untuk dikirim balik ke layar dashboard
      generatedArticles.push(`KATA KUNCI: ${kw}\n\n${aiText}`);
    } catch(e) {
      generatedArticles.push(`KATA KUNCI: ${kw}\n[GAGAL DIGENERATE: ${e.message}]`);

  // Mengembalikan array berisi teks langsung ke frontend HTML
  return generatedArticles;
}

  return { base64: base64, fileName: fileName };

    </div>
  `;

  let blob = Utilities.newBlob(htmlTemplate, MimeType.HTML).getAs(MimeType.PDF).setName(fileName);
  let base64 = Utilities.base64Encode(blob.getBytes());
  
  try {
     if (config.driveId) {
        let folder = DriveApp.getFolderById(config.driveId);
        folder.createFile(blob);
     }

  logActivity('Cetak Quotation', 'Surat Penawaran B2B: ' + quoId + ' untuk ' + name, 'Admin');
  return { base64: base64, fileName: fileName };

function getBarcodeBase64(text) {
  if (!text) return "";
  try {
    // PERBAIKAN: Tambahkan scaleX=3 untuk melebarkan barcode ke samping
    var url = "https://bwipjs-api.metafloor.com/?bcid=code128&text=" + encodeURIComponent(text) + "&scaleX=3&scaleY=1";
    var response = UrlFetchApp.fetch(url, {muteHttpExceptions: true});
    
    if (response.getResponseCode() === 200) {
      var blob = response.getBlob();
      return "data:image/png;base64," + Utilities.base64Encode(blob.getBytes());
    }
  } catch(e) {}
  return "";
}

function updateCustomerDB(name, phone, amount) {
  if(!phone) return;
  var cleanPhone = String(phone).replace(/[^0-9]/g, '');
  if(cleanPhone.startsWith('0')) cleanPhone = '62' + cleanPhone.substring(1);
  
  var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  var sheet = ss.getSheetByName('Customers');
  if (!sheet) {
    sheet = ss.insertSheet('Customers');
    sheet.appendRow(['Nama Pelanggan', 'Nomor WA', 'Total Belanja', 'Jumlah Order', 'Tanggal Terakhir']);
    sheet.getRange('A1:E1').setFontWeight('bold');
    sheet.hideSheet(); 
  }
  
  var data = sheet.getDataRange().getValues();
  var foundRow = -1;
  var currentTotal = 0;
  var currentCount = 0;
  
  for(var i=1; i<data.length; i++){
    if(String(data[i][1]) === cleanPhone) {
      foundRow = i + 1;
      currentTotal = parseFloat(data[i][2]) || 0;
      currentCount = parseInt(data[i][3]) || 0;
      break;
    }
  }
  
  var today = new Date().toLocaleDateString('id-ID');
  var orderAmount = parseFloat(amount) || 0;
  
  if(foundRow > -1) {
    sheet.getRange(foundRow, 3).setValue(currentTotal + orderAmount);
    sheet.getRange(foundRow, 4).setValue(currentCount + 1);
    sheet.getRange(foundRow, 5).setValue(today);
  } else {
    sheet.appendRow([name, cleanPhone, orderAmount, 1, today]);
  }
}

function syncCustomersFromOrders() {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var orderSheet = ss.getSheetByName('Orders');
    if(!orderSheet) return "Sheet Orders belum ada.";
    
    var orders = orderSheet.getDataRange().getValues();
    if(orders.length <= 1) return "Belum ada data pesanan untuk disinkronisasi.";
    
    // Map untuk menampung agregasi berdasarkan nomor WA
    var custMap = {}; 
    
    for(var i=1; i<orders.length; i++) {
      var name = orders[i][2];
      var phone = String(orders[i][3]).replace(/[^0-9]/g, '');
      if(!phone) continue;
      if(phone.startsWith('0')) phone = '62' + phone.substring(1);
      
      var amount = parseFloat(orders[i][6]) || 0;
      var status = orders[i][8];
      var dateStr = String(orders[i][1]); // Tanggal order
      
      // Abaikan pesanan yang dibatalkan
      if(status === 'Dibatalkan') continue;
      
      if(!custMap[phone]) {
        custMap[phone] = { name: name, total: 0, count: 0, lastOrder: dateStr };
      }
      custMap[phone].total += amount;
      custMap[phone].count += 1;
      custMap[phone].lastOrder = dateStr; 
    }
    
    // Tulis ulang ke sheet Customers
    var custSheet = ss.getSheetByName('Customers');
    if(!custSheet) {
      custSheet = ss.insertSheet('Customers');
    }
    custSheet.clear();
    custSheet.appendRow(['Nama Pelanggan', 'Nomor WA', 'Total Belanja', 'Jumlah Order', 'Tanggal Terakhir']);
    custSheet.getRange('A1:E1').setFontWeight('bold');
    
    var rowsToInsert = [];
    for(var key in custMap) {
       var c = custMap[key];
       rowsToInsert.push([c.name, key, c.total, c.count, c.lastOrder]);
    }
    
    if(rowsToInsert.length > 0) {
      custSheet.getRange(2, 1, rowsToInsert.length, 5).setValues(rowsToInsert);
    }
    
    logActivity('Sinkronisasi CRM', 'Merekap ulang data pelanggan dari Orders', 'Admin');
    return "✅ Berhasil menyinkronkan " + rowsToInsert.length + " pelanggan setia dari riwayat pesanan!";
  } catch(e) {
    throw new Error(e.message);
  }
}

function getGA4Data() {
  const settings = getSettingsData();
  // Menyaring hanya angka (GA4 Data API butuh 'Property ID', bukan 'Measurement ID' yang berawalan G-)
  const propertyIdStr = String(settings.ga).replace(/[^0-9]/g, ''); 
  
  if(!propertyIdStr || propertyIdStr.length < 5) return { success: false, configured: false };
  
  try {
    const propertyId = 'properties/' + propertyIdStr;
    const request = AnalyticsData.newRunReportRequest();
    
    // Tarik 3 Metrik Utama: Pengunjung Aktif, Sesi, dan Tayangan Halaman
    request.metrics = [
      { name: 'activeUsers' },
      { name: 'sessions' },
      { name: 'screenPageViews' }
    ];
    
    // Dimensi waktu: 30 Hari Terakhir
    request.dateRanges = [{ startDate: '30daysAgo', endDate: 'today' }];
    
    const report = AnalyticsData.Properties.runReport(request, propertyId);
    
    if (!report.rows || report.rows.length === 0) {
      return { success: true, configured: true, users: 0, sessions: 0, views: 0 };
    }
    
    const row = report.rows[0];
    return {
      success: true, configured: true,
      users: row.metricValues[0].value,
      sessions: row.metricValues[1].value,
      views: row.metricValues[2].value
    };
  } catch (e) {
    return { success: false, configured: true, error: e.message };
  }
}

function syncToMerchantCenter(p) {
  const settings = getSystemConfig();
  const mId = settings.merchantId ? settings.merchantId.toString().replace(/\s+/g, '') : '';
  if(!mId || mId.length < 5) return; 

  let safeMainImgUrl = "";
  try {
    let imgs = JSON.parse(p.img);
    if(imgs.length > 0) safeMainImgUrl = getSafeImgUrl(imgs[0]);
  } catch(e) { 
    safeMainImgUrl = getSafeImgUrl(p.img);
  }
  if(!safeMainImgUrl) safeMainImgUrl = settings.storeLogo;

  let productUrl = p.canonical ? p.canonical : settings.storeUrl + "/p/" + p.slug + ".html";
  
  let finalPrice = parseFloat(p.price);
  if (p.discount && parseFloat(p.discount) > 0 && parseFloat(p.discount) < finalPrice) {
     finalPrice = parseFloat(p.discount);
  }

  const productResource = {
    offerId: p.sku || p.slug,
    title: p.name,
    description: p.desc ? p.desc.replace(/<[^>]+>/g, '').substring(0, 4900) : p.name,
    link: productUrl,
    imageLink: safeMainImgUrl,
    contentLanguage: 'id',
    targetCountry: 'ID',
    channel: 'online',
    availability: (p.availability && p.availability.toString().toLowerCase() === 'habis') ? 'out of stock' : 'in stock',
    condition: 'new',
    price: { value: finalPrice.toString(), currency: 'IDR' }
  };

  try {
    ShoppingContent.Products.insert(productResource, mId);
    logActivity('Google Merchant API', 'Sinkronisasi produk berhasil: ' + p.name, 'Sistem');
  } catch(e) {
    logActivity('Google Merchant API Error', e.message, 'Sistem');
  }
}

function saveStaffData(staffData) {
  try {
    const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName('Users');
    const data = sheet.getDataRange().getValues();
    const headers = data[0];
    let isNew = true;
    
    // Hash password jika diisi (jika kosong saat edit, berarti password tidak diubah)
    let passToSave = staffData.password ? hashPassword(staffData.password) : "";

    // Cek apakah ini edit staf yang sudah ada
    for(let i = 1; i < data.length; i++) {
      if(data[i][headers.indexOf('Username')] == staffData.username) {
        isNew = false;
        sheet.getRange(i+1, headers.indexOf('Nama')+1).setValue(staffData.nama);
        sheet.getRange(i+1, headers.indexOf('Role')+1).setValue(staffData.role);
        sheet.getRange(i+1, headers.indexOf('Foto Profil')+1).setValue(staffData.foto);
        if(passToSave !== "") {
           sheet.getRange(i+1, headers.indexOf('Password')+1).setValue(passToSave);
        }
        logActivity('Manajemen Staff', 'Memperbarui data staf: ' + staffData.username, 'Sistem');
        break;
      }
    }
    
    // Jika staf baru
    if(isNew) {
      if(passToSave === "") return {success: false, message: "Password wajib diisi untuk akun baru!"};
      let newRow = [];
      for(let j = 0; j < headers.length; j++) {
         if(headers[j] == 'Nama') newRow.push(staffData.nama);
         else if(headers[j] == 'Username') newRow.push(staffData.username);
         else if(headers[j] == 'Password') newRow.push(passToSave);
         else if(headers[j] == 'Role') newRow.push(staffData.role);
         else if(headers[j] == 'Foto Profil') newRow.push(staffData.foto);
         else newRow.push('');
      }
      sheet.appendRow(newRow);
      logActivity('Manajemen Staff', 'Menambahkan staf baru: ' + staffData.username, 'Sistem');
    }
    return {success: true, message: 'Data akun berhasil disimpan!'};
  } catch(e) { return {success: false, message: e.toString()}; }
}

function searchLocation(keyword) {
  const config = getSettingsData();
  const apiKey = config.rajaOngkirApi;
  if (!apiKey || apiKey.length < 5) return [];

  // Kita asumsikan menggunakan akun RajaOngkir tipe PRO/Basic untuk mendukung pencarian kecamatan (subdistrict)
  // Untuk versi Starter, endpointnya adalah /city
  try {
    const url = "https://pro.rajaongkir.com/api/subdistrict"; // Pastikan akun PRO/Basic jika butuh kecamatan
    // Jika menggunakan akun Starter, ubah ke: "https://api.rajaongkir.com/starter/city"
    
    // Namun karena RajaOngkir API tidak punya endpoint text-search murni (semua diload lalu difilter manual)
    // Di sini kita load data kota saja (Starter kompatibel) jika kita mau fallback.
    // Untuk efisiensi, mari gunakan endpoint city Starter sebagai fallback jika gagal
    
    // Untuk demo/implementasi aman kita gunakan Starter City Endpoint karena paling umum
    const options = {
      "method": "get",
      "headers": { "key": apiKey },
      "muteHttpExceptions": true
    };
    
    // Mencari kota
    let res = UrlFetchApp.fetch("https://api.rajaongkir.com/starter/city", options);
    if(res.getResponseCode() == 200) {
      let data = JSON.parse(res.getContentText()).rajaongkir.results;
      // Filter berdasarkan keyword
      let filtered = data.filter(city => city.city_name.toLowerCase().includes(keyword.toLowerCase()));
      return filtered.map(city => ({
        id: city.city_id,
        name: city.type + " " + city.city_name + ", " + city.province
      })).slice(0, 10); // Ambil 10 teratas
    } else {
      // Jika error (misalnya API key salah), kembalikan kosong
      return [];
    }
  } catch(e) {
    return [];
  }
}

function checkShippingCost(destinationId, weightGram) {
  const config = getSettingsData();
  const apiKey = config.rajaOngkirApi;
  if (!apiKey || apiKey.length < 5) return {success: false, message: "API Key RajaOngkir belum diatur."};

  try {
    const originCityId = config.storeCityId || "501"; // ID Kota Yogyakarta sebagai contoh default jika tidak disetting
    
    const payload = {
      "origin": originCityId, // Harus angka (ID Kota asal)
      "destination": destinationId,
      "weight": weightGram || 1000,
      "courier": "jne" // Default JNE. Pada sistem nyata bisa ditambahkan J&T, POS, dll.
    };

    const options = {
      "method": "post",
      "headers": { "key": apiKey, "content-type": "application/x-www-form-urlencoded" },
      "payload": payload,
      "muteHttpExceptions": true
    };

    let res = UrlFetchApp.fetch("https://api.rajaongkir.com/starter/cost", options);
    if(res.getResponseCode() == 200) {
       let data = JSON.parse(res.getContentText()).rajaongkir.results[0];
       if(!data || !data.costs) return {success: false, message: "Tidak ada rute ekspedisi."};
       
       let costs = data.costs.map(c => ({
          service: data.code.toUpperCase() + " " + c.service,
          cost: c.cost[0].value,
          etd: c.cost[0].etd
       }));
       
       return {success: true, data: costs};
    } else {
       return {success: false, message: "Error dari server RajaOngkir."};
    }
  } catch(e) {
    return {success: false, message: e.message};
  }
}

function saveFlashSaleData(data) {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var sheet = ss.getSheetByName('FlashSale');
    if (!sheet) {
      sheet = ss.insertSheet('FlashSale');
      sheet.appendRow(['Nama Sesi', 'Tanggal', 'Sesi Waktu', 'Produk']);
      sheet.getRange('A1:D1').setFontWeight('bold');
    }
    sheet.appendRow([data.name, data.date, data.sessions, data.products]);
    SpreadsheetApp.flush();
    logActivity('Flash Sale', 'Jadwal flash sale baru ditambahkan: ' + data.name, 'Admin');
    return "Flash Sale berhasil disimpan!";
  } catch(e) {
    throw new Error("Gagal menyimpan flash sale: " + e.message);
  }
}

function getFlashSales() {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var sheet = ss.getSheetByName('FlashSale');
    if (!sheet) return [];
    var data = sheet.getDataRange().getValues();
    // Tambahkan nomor baris ke setiap data (untuk keperluan hapus)
    var result = [data[0]];
    for (var i = 1; i < data.length; i++) {
      var row = data[i].slice();
      row.push(i + 1); // Row number (1-indexed, skip header)
      result.push(row);
    }
    return result;
  } catch(e) {
    return [];
  }
}

function deleteFlashSaleData(rowNumber) {
  try {
    var sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName('FlashSale');
    sheet.deleteRow(rowNumber);
    SpreadsheetApp.flush();
    logActivity('Flash Sale', 'Jadwal flash sale dihapus pada baris: ' + rowNumber, 'Admin');
    return "Flash Sale berhasil dihapus!";
  } catch(e) {
    throw new Error("Gagal menghapus: " + e.message);
  }
}

// LAPORAN KEUANGAN (PESANAN SELESAI)
// ==========================================
function getLaporanKeuangan() {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var sheet = ss.getSheetByName('Orders');
    if (!sheet) return [];
    var data = sheet.getDataRange().getValues();
    // Filter hanya pesanan berstatus "Selesai"
    var result = [data[0]]; // Header
    for (var i = 1; i < data.length; i++) {
      if (data[i][8] && data[i][8].toString().toLowerCase() === 'selesai') {
        result.push(data[i]);
      }
    }
    return result;
  } catch(e) {
    return [];
  }
}
